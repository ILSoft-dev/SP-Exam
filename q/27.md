<div class="tldr" markdown="1">
**Ресурсы** — данные интерфейса, которые хранятся **внутри исполняемого файла** (EXE или DLL) отдельно от кода: значки, курсоры, изображения, меню, диалоговые окна, строки и др.

Описываются в **файле сценария ресурсов (`.rc`)** с заголовком идентификаторов `resource.h`. Компилируются **компилятором ресурсов** `rc.exe` в двоичный файл **`.res`**, который компоновщик включает в **секцию `.rsrc`** PE-файла.

Зачем ресурсы:
- **отделение интерфейса от кода** (можно менять без перекомпиляции, есть редакторы ресурсов);
- **локализация** (отдельные DLL с ресурсами на разных языках);
- **экономия памяти** (ресурсы загружаются по требованию, их страницы разделяются и не пишутся в файл подкачки).

**Виды:** `ICON`, `CURSOR`, `BITMAP`, `MENU`, `DIALOG`/`DIALOGEX`, `STRINGTABLE`, `ACCELERATORS`, `VERSIONINFO`, `RT_MANIFEST`, `FONT`, `HTML`, **пользовательские** (`RCDATA` и свои типы).

Загрузка: `LoadIcon`, `LoadCursor`, `LoadImage`, `LoadMenu`, `DialogBox`, `LoadString`, `LoadAccelerators`. Произвольные ресурсы — `FindResource` + `LoadResource` + `LockResource`.
</div>

## Сценарий ресурсов (.rc)

```c
// resource.h
#define IDI_APP       101
#define IDR_MAINMENU  102
#define IDD_ABOUT     103
#define IDR_ACCEL     104
#define IDM_FILE_OPEN 40001
#define IDM_FILE_EXIT 40002
#define IDM_HELP_ABOUT 40003
#define IDS_HELLO     1
#define IDC_NAME      1001
```

```c
// app.rc
#include <windows.h>
#include "resource.h"

IDI_APP     ICON    "app.ico"
IDB_LOGO    BITMAP  "logo.bmp"

IDR_MAINMENU MENU
BEGIN
    POPUP "&Файл"
    BEGIN
        MENUITEM "&Открыть...\tCtrl+O", IDM_FILE_OPEN
        MENUITEM SEPARATOR
        MENUITEM "&Выход",              IDM_FILE_EXIT
    END
    POPUP "&Справка"
    BEGIN
        MENUITEM "&О программе",        IDM_HELP_ABOUT
    END
END

IDD_ABOUT DIALOGEX 0, 0, 200, 80
STYLE DS_MODALFRAME | WS_POPUP | WS_CAPTION | WS_SYSMENU
CAPTION "О программе"
FONT 9, "Segoe UI"
BEGIN
    LTEXT        "Учебное приложение", -1, 10, 10, 180, 10
    EDITTEXT     IDC_NAME, 10, 25, 180, 14
    DEFPUSHBUTTON "OK", IDOK, 75, 55, 50, 14
END

STRINGTABLE
BEGIN
    IDS_HELLO "Привет, мир!"
END

IDR_ACCEL ACCELERATORS
BEGIN
    "O", IDM_FILE_OPEN, VIRTKEY, CONTROL
END

1 VERSIONINFO
FILEVERSION 1,0,0,0
BEGIN
    BLOCK "StringFileInfo"
    BEGIN
        BLOCK "041904B0"
        BEGIN
            VALUE "FileDescription", "Учебное приложение"
            VALUE "CompanyName", "ВГТУ"
        END
    END
    BLOCK "VarFileInfo" BEGIN VALUE "Translation", 0x419, 1200 END
END
```

**Сборка:**
```text
rc app.rc                    → app.res
cl main.c app.res user32.lib → app.exe   (ресурсы попадают в секцию .rsrc)
```

В Visual Studio ресурсы редактируются визуально (Resource View). MinGW использует `windres`.

## Виды ресурсов

| Вид | Тип (RT_*) | Назначение | Загрузка / использование |
|---|---|---|---|
| **Значок** | `ICON` (`RT_GROUP_ICON`) | Значок приложения и окон, разные размеры в одном `.ico` | `LoadIcon`, `LoadImage`; `wc.hIcon` |
| **Курсор** | `CURSOR` | Указатель мыши с «горячей точкой» | `LoadCursor`, `SetCursor` |
| **Растровое изображение** | `BITMAP` | Картинки, фон, панели | `LoadBitmap`, `LoadImage` → GDI ([вопрос 29]({{ '/q/29.html' | relative_url }})) |
| **Меню** | `MENU` | Главное и контекстное меню | `LoadMenu`, `wc.lpszMenuName`, `TrackPopupMenu`. Выбор пункта → `WM_COMMAND` |
| **Диалоговое окно** | `DIALOG`, `DIALOGEX` | Шаблон окна с элементами управления | `DialogBox` (модальный), `CreateDialog` (немодальный) |
| **Таблица строк** | `STRINGTABLE` | Тексты интерфейса, удобно для локализации | `LoadString` |
| **Таблица ускорителей** | `ACCELERATORS` | Горячие клавиши → команды | `LoadAccelerators`, `TranslateAccelerator` |
| **Информация о версии** | `VERSIONINFO` | Версия, компания, описание («Свойства → Подробно») | `GetFileVersionInfo`, `VerQueryValue` |
| **Манифест** | `RT_MANIFEST` (24) | Стили Common Controls v6, уровень прав UAC, DPI | Используется загрузчиком |
| **Шрифт** | `FONT` | Встроенные шрифты | `AddFontMemResourceEx` |
| **HTML** | `HTML` | Страницы для встроенного браузера | `res://` протокол |
| **Пользовательские** | `RCDATA` или своё имя типа | Любые двоичные данные: звуки, JSON, базы | `FindResource` → `LoadResource` → `LockResource` |

## Использование ресурсов в программе

```c
HINSTANCE hInst = GetModuleHandle(NULL);

// значок и меню для класса окна
wc.hIcon        = LoadIcon(hInst, MAKEINTRESOURCE(IDI_APP));
wc.lpszMenuName = MAKEINTRESOURCE(IDR_MAINMENU);

// строка
WCHAR buf[128];
LoadStringW(hInst, IDS_HELLO, buf, 128);

// ускорители в цикле сообщений
HACCEL hAcc = LoadAccelerators(hInst, MAKEINTRESOURCE(IDR_ACCEL));
while (GetMessage(&msg, NULL, 0, 0)) {
    if (!TranslateAccelerator(hwnd, hAcc, &msg)) {
        TranslateMessage(&msg);
        DispatchMessage(&msg);
    }
}

// обработка команд меню
case WM_COMMAND:
    switch (LOWORD(wParam)) {
    case IDM_HELP_ABOUT: DialogBox(hInst, MAKEINTRESOURCE(IDD_ABOUT), hwnd, AboutProc); break;
    case IDM_FILE_EXIT:  DestroyWindow(hwnd); break;
    }
    return 0;
```

**Процедура диалога:**
```c
INT_PTR CALLBACK AboutProc(HWND hDlg, UINT msg, WPARAM wParam, LPARAM lParam) {
    switch (msg) {
    case WM_INITDIALOG:
        SetDlgItemTextW(hDlg, IDC_NAME, L"Студент");
        return TRUE;
    case WM_COMMAND:
        if (LOWORD(wParam) == IDOK || LOWORD(wParam) == IDCANCEL) {
            EndDialog(hDlg, LOWORD(wParam));
            return TRUE;
        }
    }
    return FALSE;      // не обработано — стандартная обработка
}
```

**Произвольный ресурс:**
```c
HRSRC   hr   = FindResource(hInst, MAKEINTRESOURCE(IDR_DATA), RT_RCDATA);
HGLOBAL hg   = LoadResource(hInst, hr);
void   *data = LockResource(hg);      // указатель на данные в образе модуля
DWORD   size = SizeofResource(hInst, hr);
```

## Идентификация и локализация

- Ресурс идентифицируется **типом**, **именем или числовым ID** и **языком**. `MAKEINTRESOURCE(id)` превращает число в «псевдоуказатель».
- Можно хранить один ресурс на нескольких языках или вынести ресурсы в **ресурсную DLL** (MUI-файлы Windows: `*.dll.mui`) и загружать через `LoadLibraryEx(..., LOAD_LIBRARY_AS_DATAFILE)`.
- Редактировать ресурсы готовых EXE можно функциями `BeginUpdateResource` / `UpdateResource` / `EndUpdateResource` или утилитами (Resource Hacker).
